  // The right option is not always the first listed, not always the longest (the shown order is also scrambled per question by the component).
  const choices = items.filter((q) => q.type === "choice");
  const longest = choices.filter((q) => { const L = q.options.map((o) => o.text.length); return q.options.find((o) => o.correct).text.length === Math.max(...L); });
  if (longest.length > Math.floor(choices.length / 2)) throw new Error(`9.5: the correct option is the longest in ${longest.length} of ${choices.length} choice questions: ${longest.map((q) => q.id).join(", ")}`);
